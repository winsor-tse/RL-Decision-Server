import minari
from Custom_enviornments.Test_Env.Env_16_BC import Env16BC


raw_env = Env16BC()
env = minari.DataCollector(raw_env, record_infos=True)
max_steps = 500

env.reset(options={"minari_autoseed": False})
for step in range(max_steps):
    action = raw_env.next_action()
    obs, rew, terminated, truncated, info = env.step(action)
    if (terminated or truncated) and step + 1 < max_steps:
        env.reset(options={"minari_autoseed": False})

dataset = env.create_dataset(
    "env16/BC-v0",
    algorithm_name="human behavior cloning demonstrations",
)
